// Fungsi "strava" untuk RunUp. Client Secret hanya ada di Secrets Supabase, tidak di kode aplikasi.
const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};
const json = (b: unknown, s = 200) =>
  new Response(JSON.stringify(b), { status: s, headers: { ...CORS, 'Content-Type': 'application/json' } });

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: CORS });
  try {
    // hanya pengguna RunUp yang sudah masuk
    const auth = req.headers.get('Authorization') || '';
    const key = Deno.env.get('SUPABASE_ANON_KEY') || req.headers.get('apikey') || '';
    const u = await fetch(Deno.env.get('SUPABASE_URL') + '/auth/v1/user', { headers: { Authorization: auth, apikey: key } });
    if (!u.ok) return json({ error: 'Belum masuk akun. Masuk dulu di aplikasi.' }, 401);

    const ID = Deno.env.get('STRAVA_CLIENT_ID'), SECRET = Deno.env.get('STRAVA_CLIENT_SECRET');
    if (!ID || !SECRET) return json({ error: 'STRAVA_CLIENT_ID atau STRAVA_CLIENT_SECRET belum diisi di Secrets Supabase.' });

    const b = await req.json().catch(() => ({}));
    if (b.aksi === 'info') return json({ client_id: ID });

    const token = async (p: Record<string, string>) => {
      const r = await fetch('https://www.strava.com/oauth/token', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ client_id: ID, client_secret: SECRET, ...p }),
      });
      const d = await r.json();
      if (!r.ok) throw new Error('Strava menolak (' + r.status + '). Coba hubungkan ulang Strava di menu Profil.');
      return d;
    };

    if (b.aksi === 'tukar') {
      const d = await token({ code: String(b.code || ''), grant_type: 'authorization_code' });
      const a = d.athlete || {};
      return json({ refresh_token: d.refresh_token, atlet: [a.firstname, a.lastname].filter(Boolean).join(' ') });
    }

    if (b.aksi === 'aktivitas') {
      const d = await token({ refresh_token: String(b.refresh_token || ''), grant_type: 'refresh_token' });
      const out: unknown[] = [];
      for (let page = 1; page <= 3; page++) {
        const r = await fetch(`https://www.strava.com/api/v3/athlete/activities?after=${Math.floor(Number(b.after) || 0)}&per_page=100&page=${page}`,
          { headers: { Authorization: 'Bearer ' + d.access_token } });
        const a = await r.json();
        if (!r.ok) throw new Error('Gagal mengambil aktivitas Strava (' + r.status + ').');
        out.push(...a.map((x: any) => ({
          id: String(x.id), nama: x.name, jenis: x.sport_type || x.type,
          mulai: x.start_date_local, mulai_utc: x.start_date,
          jarak: x.distance, gerak: x.moving_time, maks: x.max_speed, hr: x.average_heartrate || null,
        })));
        if (a.length < 100) break;
      }
      return json({ refresh_token: d.refresh_token, aktivitas: out });
    }

    if (b.aksi === 'putus') {
      const d = await token({ refresh_token: String(b.refresh_token || ''), grant_type: 'refresh_token' });
      await fetch('https://www.strava.com/oauth/deauthorize', { method: 'POST', headers: { Authorization: 'Bearer ' + d.access_token } });
      return json({ ok: true });
    }
    return json({ error: 'Perintah tidak dikenal.' }, 400);
  } catch (e) {
    return json({ error: String((e as Error).message || e) });
  }
});
